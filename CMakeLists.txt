cmake_minimum_required(VERSION 3.20)
project(LivingWorld VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(livingworld SHARED sim/world.cpp sim/social.cpp sim/perception.cpp sim/api.cpp)
target_include_directories(livingworld PUBLIC sim)
add_executable(livingworld_native_tests tests/native.cpp sim/world.cpp sim/social.cpp sim/perception.cpp)
target_include_directories(livingworld_native_tests PRIVATE sim)
add_library(livingworld_policy SHARED inference/runtime.cpp)
target_include_directories(livingworld_policy PUBLIC inference)
add_executable(livingworld_inference_tests tests/inference_native.cpp inference/runtime.cpp)
target_include_directories(livingworld_inference_tests PRIVATE inference)
add_library(livingworld_social SHARED inference/social_runtime.cpp)
target_include_directories(livingworld_social PUBLIC inference)
add_executable(livingworld_social_tests tests/social_inference_native.cpp inference/social_runtime.cpp)
target_include_directories(livingworld_social_tests PRIVATE inference)
add_library(livingworld_social_v2 SHARED inference/social_runtime.cpp)
target_include_directories(livingworld_social_v2 PUBLIC inference)
target_compile_definitions(livingworld_social_v2 PRIVATE LW_SOCIAL_VERSION=2)
add_executable(livingworld_social_v2_tests tests/social_inference_native.cpp inference/social_runtime.cpp)
target_include_directories(livingworld_social_v2_tests PRIVATE inference)
target_compile_definitions(livingworld_social_v2_tests PRIVATE LW_SOCIAL_VERSION=2)
if(MSVC)
  target_compile_options(livingworld PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_native_tests PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_policy PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_inference_tests PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_social PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_social_tests PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_social_v2 PRIVATE /W4 /permissive-)
  target_compile_options(livingworld_social_v2_tests PRIVATE /W4 /permissive-)
else()
  target_compile_options(livingworld PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_native_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_policy PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_inference_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_social PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_social_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_social_v2 PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(livingworld_social_v2_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native COMMAND livingworld_native_tests)
add_test(NAME inference COMMAND livingworld_inference_tests)
add_test(NAME social_inference COMMAND livingworld_social_tests)
add_test(NAME social_inference_v2 COMMAND livingworld_social_v2_tests)
